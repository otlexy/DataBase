SELECT "ТипРаботы", "ЦенаЧаса"
FROM "ТипыРабот"

SELECT "Фамилия", "Адрес"
FROM "Сотрудник"
WHERE "Адрес" LIKE '%Ульяновск%'
   OR "Адрес" LIKE '%Самара%';


SELECT
    "ВыполненнаяРабота"."id_ВыполненнойРаботы",
    "ТипыРабот"."ТипРаботы",
    "ВыполненнаяРабота"."Кол-воЧасов",
    "ТипыРабот"."ЦенаЧаса",
    "ВыполненнаяРабота"."Кол-воЧасов" * "ТипыРабот"."ЦенаЧаса" AS "Стоимость"
FROM "ВыполненнаяРабота", "ТипыРабот"
WHERE "ВыполненнаяРабота"."id_ТипаРаботы" = "ТипыРабот"."id_ТипаРаботы";


SELECT
    "Сотрудник"."Фамилия",
    "ТипыРабот"."ТипРаботы",
    "ВыполненнаяРабота"."ДатаВыполнения"
FROM "Сотрудник", "ВыполненнаяРабота", "ТипыРабот"
WHERE "Сотрудник"."№ТК" = "ВыполненнаяРабота"."№ТК"
  AND "ТипыРабот"."id_ТипаРаботы" = "ВыполненнаяРабота"."id_ТипаРаботы";


SELECT
    COUNT(*)            AS "Всего операций",
    SUM("СуммаНачисления") AS "Общая сумма",
    AVG("СуммаНачисления") AS "Средняя сумма",
    MIN("СуммаНачисления") AS "Минимум",
    MAX("СуммаНачисления") AS "Максимум"
FROM "НачислениеЗП";


--то есть, из таблицы выполненнаяРабота мы ищем совпадения по id 
--из таблицы ТипыРабот, эти совпадения, по возможности группируем, 
--и дальше count и sum
SELECT
    "ТипыРабот"."ТипРаботы",
    COUNT(*)                  AS "Количество выполнений",
    SUM("ВыполненнаяРабота"."Кол-воЧасов") AS "Всего часов"
FROM "ВыполненнаяРабота"
INNER JOIN "ТипыРабот"
    ON "ВыполненнаяРабота"."id_ТипаРаботы" = "ТипыРабот"."id_ТипаРаботы"
GROUP BY "ТипыРабот"."ТипРаботы";


SELECT "Фамилия", "Дата_Устройства"

-- Найти сотрудников с максимальной зарплатой
SELECT "Фамилия", "СуммаНачисления"
FROM "Сотрудник"
INNER JOIN "НачислениеЗП" ON "Сотрудник"."№ТК" = "НачислениеЗП"."№ТК"
WHERE "СуммаНачисления" = (SELECT MAX("СуммаНачисления") FROM "НачислениеЗП");

-- Найти сотрудников, у которых есть начисления за 2023 год
SELECT "Фамилия"
FROM "Сотрудник"
WHERE "№ТК" IN ( --проверка есть ли номер ТК во вло
    SELECT "№ТК"
    FROM "НачислениеЗП"
    WHERE "ДатаНачисления" BETWEEN '2023-01-01' AND '2023-12-31');

-- Для каждого сотрудника — сколько у него работ дороже 1000 руб/час
SELECT
    "Фамилия",
    (SELECT COUNT(*)
     FROM "ВыполненнаяРабота"
     INNER JOIN "ТипыРабот" 
         ON "ВыполненнаяРабота"."id_ТипаРаботы" = "ТипыРабот"."id_ТипаРаботы"
     WHERE "ВыполненнаяРабота"."№ТК" = "Сотрудник"."№ТК" 
       AND "ТипыРабот"."ЦенаЧаса" > 1000) AS "Дорогих работ"
FROM "Сотрудник";






-- Простое нематериализованное
-- (фамилия, телефон и название должности каждого сотрудника)
CREATE VIEW "View_СотрудникиСДолжностями" AS
SELECT
    "Сотрудник"."№ТК",
    "Сотрудник"."Фамилия",
    "Сотрудник"."Телефон",
    "Должность"."НазваниеДолжности"
FROM "Сотрудник"
INNER JOIN "Должность" ON "Сотрудник"."id_Должности" = "Должность"."id_Должности";

--Материализованное неизменяемое 
--(для каждого сотрудника — общую сумму начислений)
CREATE MATERIALIZED VIEW "MView_СуммаЗППоСотрудникам" AS
SELECT
    "Сотрудник"."№ТК",
    "Сотрудник"."Фамилия",
    SUM("НачислениеЗП"."СуммаНачисления") AS "ОбщаяСуммаЗП"
FROM "Сотрудник"
INNER JOIN "НачислениеЗП" ON "Сотрудник"."№ТК" = "НачислениеЗП"."№ТК"
GROUP BY "Сотрудник"."№ТК", "Сотрудник"."Фамилия";
